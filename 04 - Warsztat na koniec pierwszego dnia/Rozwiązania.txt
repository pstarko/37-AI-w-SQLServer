WZ_TELEFON = re.compile(r"(?<!\d)(?:\+48[\s-]?)?\d{3}[\s-]?\d{3}[\s-]?\d{3}(?!\d)")
 
def maskuj(tekst):
    licznik = 0
    for wzorzec, znacznik in [(WZ_EMAIL, "[EMAIL]"), (WZ_PESEL, "[PESEL]"), (WZ_TELEFON, "[TELEFON]")]:
        tekst, n = wzorzec.subn(znacznik, tekst)
        licznik += n
    return tekst, licznik

df["TrescCzysta"] = df["Tresc"].apply(lambda t: maskuj(t)[0])

TRUNCATE TABLE BOK.ZgloszeniaCzyste;   -- pozwala bezpiecznie uruchomić zadanie ponownie
 
INSERT INTO BOK.ZgloszeniaCzyste (ZgloszenieID, TrescCzysta, LiczbaMaskowan, LiczbaZnakow, CzyPilne)
EXEC sp_execute_external_script
     @language     = N'Python',
     @script       = N'
import re
 
WZ_EMAIL   = re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+")
WZ_PESEL   = re.compile(r"(?<!\d)\d{11}(?!\d)")
WZ_TELEFON = re.compile(r"(?<!\d)(?:\+48[\s-]?)?\d{3}[\s-]?\d{3}[\s-]?\d{3}(?!\d)")
SLOWA_PILNE = ["niebezpieczn", "pilne", "wypadek", "nie działa", "przestają działać"]
 
def maskuj(tekst):
    licznik = 0
    for wzorzec, znacznik in [(WZ_EMAIL, "[EMAIL]"), (WZ_PESEL, "[PESEL]"), (WZ_TELEFON, "[TELEFON]")]:
        tekst, n = wzorzec.subn(znacznik, tekst)
        licznik += n
    return tekst, licznik
 
df = InputDataSet.copy()
wyniki = df["Tresc"].apply(maskuj)
df["TrescCzysta"]    = wyniki.apply(lambda w: w[0])
df["LiczbaMaskowan"] = wyniki.apply(lambda w: w[1])
df["LiczbaZnakow"]   = df["TrescCzysta"].str.len()
df["CzyPilne"]       = df["Tresc"].str.lower().apply(lambda t: int(any(s in t for s in SLOWA_PILNE)))
OutputDataSet = df[["ZgloszenieID", "TrescCzysta", "LiczbaMaskowan", "LiczbaZnakow", "CzyPilne"]]
',
     @input_data_1 = N'SELECT ZgloszenieID, Tresc FROM BOK.Zgloszenia';
 
SELECT * FROM BOK.ZgloszeniaCzyste ORDER BY ZgloszenieID;
