     @language     = N'Python',

import re
 
WZ_EMAIL   = re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+")
WZ_PESEL   = re.compile(r"(?<!\d)\d{11}(?!\d)")
WZ_TELEFON = re.compile(r"(?<!\d)(?:\+48[\s-]?)?\d{3}[\s-]?\d{3}[\s-]?\d{3}(?!\d)")
SLOWA_PILNE = ["niebezpieczn", "pilne", "wypadek", "nie działa", "przestają działać"]
 
def maskuj(tekst):
    licznik = 0
    for wzorzec, znacznik in [(WZ_EMAIL, "[EMAIL]"), (WZ_PESEL, "[PESEL]"), (WZ_TELEFON, "[TELEFON]")]:
        tekst, n = wzorzec.subn(znacznik, tekst)
        licznik += n
    return tekst, licznik
 
df = InputDataSet.copy()
wyniki = df["Tresc"].apply(maskuj)
df["TrescCzysta"]    = wyniki.apply(lambda w: w[0])
df["LiczbaMaskowan"] = wyniki.apply(lambda w: w[1])
df["LiczbaZnakow"]   = df["TrescCzysta"].str.len()
df["CzyPilne"]       = df["Tresc"].str.lower().apply(lambda t: int(any(s in t for s in SLOWA_PILNE)))
OutputDataSet = df[["ZgloszenieID", "TrescCzysta", "LiczbaMaskowan", "LiczbaZnakow", "CzyPilne"]]

     @input_data_1 = N'SELECT ZgloszenieID, Tresc FROM BOK.Zgloszenia';

DECLARE @tresc NVARCHAR(2000) =
        (SELECT TrescCzysta FROM BOK.ZgloszeniaCzyste WHERE ZgloszenieID = 3);
 
DECLARE @payload NVARCHAR(MAX) = JSON_OBJECT(
        'model': 'gpt-4o-mini',
        'temperature': 0,
        'messages': JSON_ARRAY(
            JSON_OBJECT('role': 'system', 'content': N'Przypisz zgłoszenie do jednej kategorii: Reklamacja, Dostawa, Pytanie o produkt, Serwis, Inne. Odpowiedz tylko nazwą kategorii.'),
            JSON_OBJECT('role': 'user',   'content': @tresc)));
 
DECLARE @response NVARCHAR(MAX), @ret INT;
 
EXEC @ret = sys.sp_invoke_external_rest_endpoint
     @url        = N'https://api.openai.com/v1/chat/completions',
     @method     = 'POST',
     @credential = [https://api.openai.com],
     @payload    = @payload,
     @response   = @response OUTPUT;
 
SELECT @ret                                                         AS KodPowrotu,       -- 0 = OK
       JSON_VALUE(@response, '$.result.choices[0].message.content') AS Kategoria,        -- Dostawa
       JSON_VALUE(@response, '$.result.usage.total_tokens')         AS Tokeny,
       JSON_VALUE(@response, '$.response.status.http.code')         AS KodHTTP,          -- 200
       @response                                                    AS PelnaOdpowiedz;

CREATE TABLE BOK.ProduktyWektory
(
    ProductModelID INT            NOT NULL CONSTRAINT PK_ProduktyWektory PRIMARY KEY,
    Nazwa          NVARCHAR(50)   NOT NULL,
    Opis           NVARCHAR(400)  NOT NULL,
    Wektor         VECTOR(1536)   NOT NULL
);
GO
INSERT INTO BOK.ProduktyWektory (ProductModelID, Nazwa, Opis, Wektor)
SELECT pm.ProductModelID,
       pm.Name,
       pd.Description,
       AI_GENERATE_EMBEDDINGS(CONCAT(pm.Name, N': ', pd.Description) USE MODEL EmbeddingiOpenAI)
FROM Production.ProductModel AS pm
JOIN Production.ProductModelProductDescriptionCulture AS pmc
     ON pmc.ProductModelID = pm.ProductModelID
    AND pmc.CultureID = N'en'
JOIN Production.ProductDescription AS pd
     ON pd.ProductDescriptionID = pmc.ProductDescriptionID;
 
SELECT COUNT(*) AS LiczbaProduktow FROM BOK.ProduktyWektory;
